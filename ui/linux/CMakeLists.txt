cmake_minimum_required(VERSION 3.22)
project(RustClipSyncLinux VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network)
qt_standard_project_setup()
qt_add_executable(rustclipsync main.cpp Backend.cpp Backend.h)
qt_add_resources(rustclipsync resources PREFIX "/" FILES Main.qml HistoryPane.qml SensitiveWindow.qml ../../assets/rustclipsync.png)
target_link_libraries(rustclipsync PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network)
target_compile_options(rustclipsync PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS rustclipsync RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(NOT AGENT_BINARY)
  set(AGENT_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release/rustclipsync-agent")
endif()
install(PROGRAMS "${AGENT_BINARY}" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/rustclipsync)
install(FILES ../../packaging/linux/io.github.infinifar.rustclipsync.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ../../assets/rustclipsync.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps RENAME io.github.infinifar.rustclipsync.png)
install(FILES ../../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/rustclipsync-desktop)
enable_testing()
add_test(NAME linux-ui-smoke COMMAND rustclipsync --smoke-test)
set_tests_properties(linux-ui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
